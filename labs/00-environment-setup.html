10d. RAG skeleton

def rag(question):
    search_results = search(question)          # 1. Retrieval
    user_prompt = build_prompt(question, search_results)  # 2. Prompt
    return llm(user_prompt)                    # 3. Generation

        print(f"iteration #{it}...")
        has_function_calls = False

        response = openai_client.responses.create(
            model=model, input=messages, tools=[search_tool])
        messages.extend(response.output)

        for item in response.output:
            if item.type == "function_call":
                print("function_call:", item.name, item.arguments)
                messages.append(make_call(item))
                has_function_calls = True
            elif item.type == "message":
                last_answer = item.content[0].text
                print("ASSISTANT:", last_answer)

        it = it + 1
        if has_function_calls == False:
            break
    return last_answer

class RAGPgVector(RAGBase):
    def __init__(self, embedder, conn, **kwargs):
        super().__init__(index=None, **kwargs)
        self.embedder = embedder
        self.conn = conn

    def search(self, query, num_results=5):
        query_str = vec_to_str(self.embedder.encode(query))
        rows = self.conn.execute("""SELECT course, section, question, answer
            FROM documents WHERE course = %s
            ORDER BY embedding <=> %s::vector LIMIT %s""",
